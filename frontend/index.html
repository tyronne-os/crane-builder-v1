// src/App.tsx — crane-it landing page
import { useState, useEffect } from 'react'
import { Hero, StatsStrip, Services } from './components'

const SLIDES = [
  { a: 'Build',  b: 'faster'  },
  { a: 'Deploy', b: 'smarter' },
  { a: 'Ship',   b: 'confidently' },
]

export default function App() {
  const [slide, setSlide] = useState(0)

  useEffect(() => {
    const t = setInterval(
      () => setSlide(s => (s + 1) % SLIDES.length),
      3200
    )
    return () => clearInterval(t)
  }, [])

  return (
    <main className="bg-ink">
      <Hero
        bgSrc="uploads/hero-bg.jpeg"
        slide={SLIDES[slide]}
        cta="Get shit done"
      />
      <StatsStrip
        items={[
          { n: '12,847', label: 'Builds shipped' },
          { n: '98.7%',  label: 'Deploy success' },
          { n: '34',     label: 'Models supported' },
          { n: '99.9%',  label: 'Uptime SLA' },
        ]}
      />
      <Services />
    </main>
  )
}
